<script>
    function triggerEnvelopeOpen() {
        const flap = document.getElementById('flap');
        const env = document.getElementById('envelope');
        const wax = document.getElementById('waxSeal');

        createHearts();

        wax.style.transform = 'translate(-50%, -50%) scale(1.3)';
        
        setTimeout(() => {
            flap.style.transform = 'rotateX(180deg)';
            flap.style.zIndex = '1';
        }, 300);

        setTimeout(() => {
            env.style.transform = 'translateY(120px) scale(0.9)';
            env.style.opacity = '0';
        }, 800);

        setTimeout(() => {
            env.style.display = 'none';
            document.getElementById('card-welcome').classList.add('active');
        }, 1200);
    }

    function showCard(cardId) {
        document.querySelectorAll('.card').forEach(card => {
            card.classList.remove('active');
        });
        document.getElementById(cardId).classList.add('active');
    }

    function correctAndNext(nextCardId) {
        createHearts();
        showCard(nextCardId);
    }

    function wrongAnswer() {
        alert("Oops! Try again handsome 😉❤️");
    }

    function createHearts() {
        const container = document.getElementById('app');
        for (let i = 0; i < 8; i++) {
            const heart = document.createElement('div');
            heart.classList.add('heart');
            heart.innerHTML = ['💖', '✨', '🎈', '🎉'][Math.floor(Math.random() * 4)];
            heart.style.left = (20 + Math.random() * 60) + '%';
            heart.style.top = (40 + Math.random() * 30) + '%';
            container.appendChild(heart);

            setTimeout(() => {
                heart.remove();
            }, 2000);
        }
    }
</script>
